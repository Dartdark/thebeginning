{=============================================================================
 NQ OVERNIGHT TICK-RSV / VIX  (long only)  - assembled by Claude 2026-10-24 from the user's specification
 'NQ Overnight Anomaly Strategy Optimized' (IS 2017-05-21 .. 2024-02-12, OOS 2024-02-13 .. 2026-07-09).
 Rebuild / tests: analysis/overnight_tick_test.py, analysis/overnight_tick_bench.py.

 RULE
   After a NEGATIVE regular session (RTH close < RTH open, 09:30-16:00 ET = 08:30-15:00 CT) that was not an
   early close (last RTH bar before 15:00 ET / 14:00 CT), go long 1 NQ at the evening reopen when
       (TickRSV <  0 and VIX close >  SMA(VIX, VIXLen))   or
       (TickRSV >= 0 and VIX close <= SMA(VIX, VIXLen))
   TickRSV = (UpTicks - DownTicks) / (UpTicks + DownTicks) summed over 15:15-16:15 ET (14:15-15:15 CT).
   Exit at 04:00 ET (03:00 CT) at market. No stop, no target (optional disaster stop below, off).
   One trade per regular session: Sunday evening uses Friday's session; a holiday evening without a new
   session does not trade again.

 CHART
   Data1: @NQ (or MNQ), 5-minute bars, CME 24-hour session, bar-close time stamps, IOG off.
          'Build Volume On' = Tick Count or Trade Volume: TickRSV uses UpTicks / DownTicks. NOTE: the test used
          aggressor buy / sell volume (The Bridge); a bar-direction proxy was much weaker (t 3.2 vs 5.2), so check
          that this chart's UpTicks / DownTicks give the same signs before trusting it (PrintDiag true).
   Data2: VIX, DAILY bars. The rule uses that day's VIX close (VIX closes 15:15 CT, before the 17:05 CT entry);
          if MultiCharts still shows yesterday's daily bar at 17:05, the result changes: check with PrintDiag.
   TIME PRESETS (set them to the chart's clock, not the code's):
     CENTRAL chart (Exchange time, our setup):  RTH 830 / 1500, RSV 1415 / 1515, Entry 1705, Exit 300, Early 1400
     EASTERN chart (the author's):               RTH 930 / 1600, RSV 1515 / 1615, Entry 1805, Exit 400, Early 1500
   Apex: the 17:05 -> 03:00 CT hold sits inside one Apex trading day (it starts 17:00 CT).
=============================================================================}

DefineDLLFunc: "kernel32.dll", int, "GetFileAttributesA", lpstr;   { peak file check (2026-11-09) }

Inputs:
    NumContracts(1),
    VIXLen(25),                     // author's pick 25; their code default was 20
    RTHStart(830), RTHEnd(1500),    // CT
    RSVStart(1415), RSVEnd(1515),   // CT, bars stamped > start and <= end
    EntryTime(1705), ExitTime(300), // CT; entry decision at the first bar stamped >= EntryTime in the evening
    EarlyCloseBefore(1400),         // RTH session whose last bar is before this = early close: no trade
    UseDisasterStop(false), DisasterMult(3), RangeLen(20),
    PrintDiag(false),
	HeartbeatTag(""),      { live guard: the chart's logger tag; empty = no heartbeat }
	UseCushion(false),      { FUNDED cushion guard (FUNDED_PLAN 'Cushion cap'; added to this file 2026-11-09): true on funded charts only }
	AccountName(""),        { empty = the account in this chart's Auto Trading gear (GetAccountID); or type it to override }
	StartBalance(50000),    { 150K: 150000 }
	TrailDD(2000),          { 50K 2000 / 150K 4000 }
	LockBalance(50100),     { 50K 50100 / 150K 150100 }
	PeakSeed(0),            { highest END-OF-DAY balance so far (new account: StartBalance); RETYPE after any restart / recompile; 0 = no entries }
	CushionRef(1500),       { full size from this cushion (50K 1500 / 150K 3000) }
	CushionMin(0.5);        { size floor of the cushion sizing }
Variables: HB_Last(-1), HB_Key(0), HB_Day(""), HB_Ver("NQ_Overnight_Tick_VIX 2026-11-09");   { code version: bump the date on every change }
Variables: CE(0), CEPrev(0), CK(0), CKLast(0), PeakEOD(0), Thr(0), Cush(0), CF(0), CQty(0);   { cushion guard }
Variables: PeakAuto(0), PKey(0), PKeyLast(0), PAcct(""), PkI(0), PkFound(0);   { peak file (2026-11-09) }

Variables:
    RTHOpen(0), RTHClose(0), RTHLast(0), InRTH(false),
    UpSum(0), DnSum(0),
    StateReady(false), StateDown(false), StateEarly(false), StateRSV(0), StateHasRSV(false), StateUsed(true),
    VIXc(0), VIXma(0), Pass(false),
    NightHi(0), NightLo(0), InNight(false), AvgRange(0), NRanges(0), kk(0),
    StopPx(0);

Arrays: Rng[100](0);

// ---- regular session: open, close, last bar, TickRSV window ----
If Time > RTHStart and Time <= RTHEnd then begin
    If InRTH = false then begin
        InRTH = true; RTHOpen = Open; UpSum = 0; DnSum = 0;
    end;
    RTHClose = Close; RTHLast = Time;
end;
If Time > RSVStart and Time <= RSVEnd then begin
    UpSum = UpSum + UpTicks; DnSum = DnSum + DownTicks;
end;
// session state is frozen once the TickRSV window has ended (15:15 CT), or at the first bar after the RTH ends
If InRTH and (Time > RSVEnd or Time < RTHStart or Date <> Date[1]) then begin
    InRTH = false;
    StateDown = RTHClose < RTHOpen;
    StateEarly = RTHLast < EarlyCloseBefore;
    StateHasRSV = UpSum + DnSum > 0;
    If StateHasRSV then StateRSV = (UpSum - DnSum) / (UpSum + DnSum);
    StateReady = true; StateUsed = false;
    If PrintDiag then
        Print("OTV ", ELDateToString(Date), " RTH o ", RTHOpen:0:2, " c ", RTHClose:0:2, " last ", RTHLast:0:0,
              " up ", UpSum:0:0, " dn ", DnSum:0:0, " rsv ", StateRSV:0:4);
end;

// ---- overnight range for the optional disaster stop (EntryTime -> ExitTime window) ----
If Time >= EntryTime or Time <= ExitTime then begin
    If InNight = false then begin InNight = true; NightHi = High; NightLo = Low; end
    else begin NightHi = MaxList(NightHi, High); NightLo = MinList(NightLo, Low); end;
end
else if InNight then begin
    InNight = false;
    For kk = RangeLen - 1 downto 1 begin Rng[kk] = Rng[kk - 1]; end;
    Rng[0] = NightHi - NightLo;
    If NRanges < RangeLen then NRanges = NRanges + 1;
    AvgRange = 0;
    For kk = 0 to NRanges - 1 begin AvgRange = AvgRange + Rng[kk]; end;
    AvgRange = AvgRange / MaxList(1, NRanges);
end;

// ---- entry: first evening bar stamped >= EntryTime, flat, state not used yet ----
If MarketPosition = 0 and StateReady and StateUsed = false and Time >= EntryTime and Time < 2359 then begin
    StateUsed = true;                                   // one decision per regular session
    VIXc = Close of data2;
    VIXma = Average(Close, VIXLen) of data2;          // computed on Data2's daily bars, not on Data1's 5-min bars
    Pass = StateDown and StateEarly = false and StateHasRSV
           and ((StateRSV < 0 and VIXc > VIXma) or (StateRSV >= 0 and VIXc <= VIXma));
    If PrintDiag then
        Print("OTV ", ELDateToString(Date), " ", Time:0:0, " down ", IFF(StateDown, 1, 0):0:0, " early ", IFF(StateEarly, 1, 0):0:0,
              " rsv ", StateRSV:0:4, " vix ", VIXc:0:2, " ma ", VIXma:0:2, " VIXdate ", ELDateToString(Date of data2),
              " enter ", IFF(Pass, 1, 0):0:0);
    CQty = NumContracts;
    { cushion guard (2026-11-09, as the plan's other funded files): size by the cushion above Apex's threshold and never
      risk more than the cushion at the stop. Real time only (backtests unchanged). Equity missing or PeakSeed 0 -> no entry. }
    if UseCushion and GetAppInfo(aiRealTimeCalc) = 1 then begin
    	CE = GetRTAccountEquity(IffString(AccountName <> "", AccountName, GetAccountID));
    	Thr = MinList(LockBalance, MaxList(PeakEOD, StartBalance, PeakSeed, PeakAuto) - TrailDD);
    	Cush = CE - Thr;
    	if CE <= 0 or Cush <= 0 or (PeakSeed <= 0 and PeakAuto <= 0) then CQty = 0
    	else begin
    		CF = MinList(1, MaxList(CushionMin, Cush / CushionRef));
    		CQty = Floor(CQty * CF + 0.000001);
    		if IFF(UseDisasterStop, DisasterMult * AvgRange, 0) > 0 then CQty = MinList(CQty, Floor(Cush / ((IFF(UseDisasterStop, DisasterMult * AvgRange, 0)) * BigPointValue)));
    	end;
    end;
    If Pass and CQty >= 1 then begin
        Buy ("OTV_L") CQty contracts next bar at market;
        If UseDisasterStop and AvgRange > 0 then StopPx = Close - DisasterMult * AvgRange;
    end;
end;

// ---- exits ----
If MarketPosition > 0 then begin
    If Time >= ExitTime and Time < EntryTime then
        Sell ("OTV_X") next bar at market
    else if UseDisasterStop and StopPx > 0 then
        Sell ("OTV_DS") next bar at StopPx stop;
end;

// Heartbeat (live guard, as the apex files 2026-10-30; added to the funded files 2026-11-05). Log only: no orders.
// Set HeartbeatTag to the chart's logger tag; empty = off (backtests unchanged).
if HeartbeatTag <> "" and GetAppInfo(aiRealTimeCalc) = 1 and LastBarOnChart then begin
	HB_Key = CurrentDate * 10000 + CurrentTime;
	if HB_Key <> HB_Last then begin
		HB_Day = NumToStr(CurrentDate + 19000000, 0);
		FileAppend("C:\TradingLogs\" + HB_Day + ".csv", "SIG," + HeartbeatTag + "," + GetSymbolName + "," + HB_Ver + ","
			+ NumToStr(MarketPosition * CurrentContracts, 0) + ",,,,,,,," + HB_Day + "," + NumToStr(CurrentTime, 0) + NewLine);
		HB_Last = HB_Key;
	end;
end;

{ Peak from the VPS (2026-11-09, user: no retyping): each day after the close the VPS writes C:\TradingLogs\peak\<account>_<the
  highest end-of-day balance so far, rounded UP to $100, capped at LockBalance + TrailDD> (live/vps/cushion_peak.ps1). Read once
  per trading day here, so PeakSeed no longer needs retyping after a restart (PeakSeed > 0 still overrides upwards). No file
  (and PeakSeed 0) = no entries, as before. Real time only. }
if UseCushion and GetAppInfo(aiRealTimeCalc) = 1 then begin
	if Time >= 1600 then PKey = Date + 1 else PKey = Date;
	PAcct = IffString(AccountName <> "", AccountName, GetAccountID);
	if PKey <> PKeyLast and PAcct <> "" then begin      { until found today: a new account's file can appear during the day }
		PkFound = 0;
		for PkI = Ceiling((LockBalance + TrailDD) / 100) downto Floor(StartBalance / 100) begin
			if PkFound = 0 and GetFileAttributesA("C:\TradingLogs\peak\" + PAcct + "_" + NumToStr(PkI * 100, 0)) <> -1 then PkFound = PkI * 100;
		end;
		if PkFound > 0 then begin PeakAuto = MaxList(PeakAuto, PkFound); PKeyLast = PKey; end;
	end;
end;

// Cushion guard tracker (2026-11-09, as the plan's other funded files): the highest END-OF-DAY account equity (Apex trails the
// threshold on it). The day rolls at 16:00 CT; the last equity read before the roll is that day's close. Real time only.
if UseCushion and GetAppInfo(aiRealTimeCalc) = 1 then begin
	CE = GetRTAccountEquity(IffString(AccountName <> "", AccountName, GetAccountID));
	if Time >= 1600 then CK = Date + 1 else CK = Date;
	if CE > 0 then begin
		if CKLast <> 0 and CK <> CKLast and CEPrev > 0 then PeakEOD = MaxList(PeakEOD, CEPrev);
		CKLast = CK; CEPrev = CE;
	end;
end;
