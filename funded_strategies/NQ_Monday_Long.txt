Inputs:
	EntryBarTime(830),      { the bar closing at 8:30 -> buy at the next bar's open }
	ExitTime(1500),
	EarlyExitTime(1145),
	UseHolidayTimes(true),
	TradeDay(1),            { 1 = Monday }
	AtrDays(15),
	StopAtrPct(0.50),
	RiskPerTrade(0),        { $ lost at the stop; 0 = trade NumContracts }
	NumContracts(1);

Variables:
	DayHi(0), DayLo(0), DayClose(0), PrevDayClose(0),
	TR(0), TrCount(0), TrIdx(0), TrSum(0), Ti(0), ATRv(0),
	StopPts(0), Qty(0), TradedToday(false), DayStartTrades(0),
	CalMonth(0), CalDay(0), CalDow(0), CalYear(0), IsEarlyClose(false), DayExit(0);

Arrays:
	TrHist[100](0);

{ ---------- Daily ATR (weekdays, calendar dates), as in Overnight Bias ---------- }
if Date <> Date[1] then begin
	if DayClose > 0 then begin
		if DayOfWeek(Date[1]) >= 1 and DayOfWeek(Date[1]) <= 5 then begin
			if PrevDayClose > 0 then
				TR = MaxList(DayHi, PrevDayClose) - MinList(DayLo, PrevDayClose)
			else
				TR = DayHi - DayLo;
			TrHist[TrIdx] = TR;
			TrIdx = Mod(TrIdx + 1, AtrDays);
			if TrCount < AtrDays then TrCount = TrCount + 1;
		end;
		PrevDayClose = DayClose;
	end;
	DayHi = High;
	DayLo = Low;
	DayStartTrades = TotalTrades;
end
else begin
	DayHi = MaxList(DayHi, High);
	DayLo = MinList(DayLo, Low);
end;
DayClose = Close;

ATRv = 0;
if TrCount >= AtrDays then begin
	TrSum = 0;
	for Ti = 0 to AtrDays - 1 begin
		TrSum = TrSum + TrHist[Ti];
	end;
	ATRv = TrSum / AtrDays;
end;

{ ---------- CME early-close Mondays ---------- }
CalMonth = Month(Date);
CalDay = DayOfMonth(Date);
CalDow = DayOfWeek(Date);
CalYear = Year(Date) + 1900;
IsEarlyClose = UseHolidayTimes and CalDow = 1 and (
	(CalMonth = 1 and (CalDay = 1 or CalDay = 2))
	or (CalMonth = 1 and CalDay >= 15 and CalDay <= 21)
	or (CalMonth = 2 and CalDay >= 15 and CalDay <= 21)
	or (CalMonth = 5 and CalDay >= 25)
	or (CalMonth = 6 and CalYear >= 2022 and (CalDay = 19 or CalDay = 20))
	or (CalMonth = 7 and (CalDay = 4 or CalDay = 5))
	or (CalMonth = 9 and CalDay <= 7)
	or (CalMonth = 12 and (CalDay = 25 or CalDay = 26)));
if IsEarlyClose then DayExit = EarlyExitTime else DayExit = ExitTime;

{ ---------- Entry: long at the 8:30 open ---------- }
TradedToday = MarketPosition <> 0 or TotalTrades > DayStartTrades;
if CalDow = TradeDay and IsEarlyClose = false and Time = EntryBarTime and MarketPosition = 0
	and TradedToday = false and ATRv > 0 then begin
	StopPts = StopAtrPct * ATRv;
	Qty = NumContracts;
	if RiskPerTrade > 0 then Qty = Floor(RiskPerTrade / (StopPts * BigPointValue));
	if Qty >= 1 then
		Buy ("Monday L") Qty contracts next bar at market;
end;

{ ---------- Exits ---------- }
SetStopContract;
if StopPts > 0 then SetStopLoss(StopPts * BigPointValue);

if MarketPosition > 0 and Time >= DayExit then
	Sell ("Monday LX") this bar on close;
