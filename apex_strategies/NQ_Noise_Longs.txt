  NQ NOISE LONGS (Bridge noise area, RTH-anchored, LONGS ONLY) - 150K EVALUATION, account #5  (Claude, 2026-11-12)
  From checks/candidates/bridge_noise_15m/NQ_Noise_RTH_Anchor.txt (The Bridge's posted code + RTH anchor + early-close fix),
  unchanged trading logic; this file adds the LIVE GUARDS and sets the evaluation defaults:
    ShortTrade false (longs only), ContractsPerTrade 3 (MNQ)
    kill switch (C:\TradingLogs\KILL, tri-state with GetLastError, as Twin_Continuation_150K.txt): no new entries; an open
      position exits at market; no entries if C:\TradingLogs is unreadable. Real time only: backtests unchanged.
    heartbeat (HeartbeatTag, HB_Ver) for the Chart Monitor / watchdog
  Evidence: TIER_LIST 'Bridge noise-area NQ 15-min' (B, longs): MC-confirmed RTH-anchored run, longs $61.5 per NQ, PF 1.27, t 5.4,
  9/9 years. Eval bench vs the live #5 book (analysis/active_scan_eval.py): + NBL 3 MNQ + LDF $1,000: fresh 2023-26 +4.1 pt
  (band +2.0/+6.5), from #5 +2.0 pt (+0.6/+3.6). The MC run was BOTH sides; send a longs-only report (ShortTrade false) for a check.
  INSTRUMENT MNQ (or NQ for the backtest)   TIMEFRAME 15 minutes   TIME ZONE Exchange (Central)   CHART CME 24 Hour session
  ========================================================================== }
DefineDLLFunc: "kernel32.dll", int, "GetFileAttributesA", lpstr;
DefineDLLFunc: "kernel32.dll", int, "GetLastError";

Inputs:
    LookbackDays(20),
    VolatilityMultiplier(1.0),
    EMALength(20),
    StopLossPoints(50),
    ContractsPerTrade(3),           // 2026-11-12: 3 MNQ on the 150K evaluation
    RewardRisk(0.5),
    LongTrade(true),
    ShortTrade(false),              // 2026-11-12: longs only (the benched version)
    MaxTradesPerDay(10),
    MaxStopsPerDay(3),
    FirstEvalTime(0900),
    LastEntryTime(1400),
    FlatSignalTime(1430),
    RTHFirstBarTime(0845),          // RTH ANCHOR: the 08:30-08:45 bar (bars stamped at their close)
    RTHLastBarTime(1500),           // RTH ANCHOR: the 14:45-15:00 bar
    UseHolidayTimes(true),          // early-close days (2026-11-05)
    EarlyLastEntry(1100),
    EarlyFlattenTime(1130),
    HeartbeatTag("");              // live guard: the chart's logger tag, e.g. NBL150-5; empty = no heartbeat

Vars:
    CalMonth(0), CalDay(0), CalDow(0), CalYear(0), IsWeekday(false), IsEarlyClose(false), DayLastEntry(0),
    TodayOpen(0),
    PrevClose(0),
    LastRTHClose(0),                // RTH ANCHOR
    CurrentHour(0),
    CurrentMinute(0),
    CurrentMinutes(0),
    FirstHour(0),
    FirstMinute(0),
    FirstMinutes(0),
    Slot(0),
    HistIndex(0),
    MoveToday(0),
    SumMove(0),
    AvgMove(0),
    UpperBase(0),
    LowerBase(0),
    UpperBand(0),
    LowerBand(0),
    EMAValue(0),
    LongTrail(0),
    ShortTrail(0),
    HistLoop(0),
    ShiftLoop(0),
    StopDollarsPerPosition(0),
    TradesToday(0),
    StopsToday(0),
    LastTotalTrades(0),
    EnoughData(false),
    KillOn(false), NoEntry(false), KillAttr(-1), KillErr(0), HB_Last(-1), HB_Key(0), HB_Day(""),
    HB_Ver("NQ_Noise_Longs 2026-11-12");

Arrays:
    MoveHistory[2200](0),
    SlotCount[22](0);

StopDollarsPerPosition = StopLossPoints * BigPointValue * ContractsPerTrade;

if ContractsPerTrade >= 1 then
begin
    SetStopContract;
    SetStopLoss(StopLossPoints * BigPointValue);
    SetProfitTarget(StopLossPoints * BigPointValue * RewardRisk);
end;

if CurrentBar = 1 then
begin
    LastTotalTrades = TotalTrades;
end;

if Date <> Date[1] then
begin
    TradesToday = 0;
    StopsToday = 0;
    LastTotalTrades = TotalTrades;
    TodayOpen = 0;                  // RTH ANCHOR: no band until today's 08:30 bar
end;

// RTH ANCHOR: day open = the 08:30 open, prior close = the last 15:00 close
if Time = RTHFirstBarTime then
begin
    TodayOpen = Open;
    PrevClose = LastRTHClose;
end;

if TotalTrades > LastTotalTrades then
begin
    if PositionProfit(1) <= -StopDollarsPerPosition then
        StopsToday = StopsToday + 1;
    LastTotalTrades = TotalTrades;
end;

EMAValue = XAverage(Close, EMALength);

CurrentHour = IntPortion(Time / 100);
CurrentMinute = Time - CurrentHour * 100;
CurrentMinutes = CurrentHour * 60 + CurrentMinute;

FirstHour = IntPortion(FirstEvalTime / 100);
FirstMinute = FirstEvalTime - FirstHour * 100;
FirstMinutes = FirstHour * 60 + FirstMinute;

// early-close days (2026-11-05, same list as NQ_Vault_Break)
CalMonth = Month(Date);
CalDay = DayOfMonth(Date);
CalDow = DayOfWeek(Date);
CalYear = Year(Date) + 1900;
IsWeekday = CalDow >= 1 and CalDow <= 5;
IsEarlyClose = UseHolidayTimes and (
	(CalMonth = 1 and ((CalDay = 1 and IsWeekday) or (CalDay = 2 and CalDow = 1)))
	or (CalMonth = 1 and CalDow = 1 and CalDay >= 15 and CalDay <= 21)
	or (CalMonth = 2 and CalDow = 1 and CalDay >= 15 and CalDay <= 21)
	or (CalMonth = 5 and CalDow = 1 and CalDay >= 25)
	or (CalMonth = 6 and CalYear >= 2022 and ((CalDay = 19 and IsWeekday)
		or (CalDay = 18 and CalDow = 5) or (CalDay = 20 and CalDow = 1)))
	or (CalMonth = 7 and (((CalDay = 3 or CalDay = 4) and IsWeekday) or (CalDay = 5 and CalDow = 1)))
	or (CalMonth = 9 and CalDow = 1 and CalDay <= 7)
	or (CalMonth = 11 and CalDow = 4 and CalDay >= 22 and CalDay <= 28)
	or (CalMonth = 11 and CalDow = 5 and CalDay >= 23 and CalDay <= 29)
	or (CalMonth = 12 and (((CalDay = 24 or CalDay = 25) and IsWeekday) or (CalDay = 26 and CalDow = 1))));
if IsEarlyClose then DayLastEntry = EarlyLastEntry else DayLastEntry = LastEntryTime;

// live guards (2026-11-12, as Twin_Continuation_150K.txt): kill switch. Real time only: backtests unchanged
KillOn = false; KillAttr = -1; KillErr = 0;
if GetAppInfo(aiRealTimeCalc) = 1 then
begin
    KillAttr = GetFileAttributesA("C:\TradingLogs\KILL");
    if KillAttr <> -1 then KillOn = true else KillErr = GetLastError;
end;
NoEntry = KillOn;
if GetAppInfo(aiRealTimeCalc) = 1 and GetFileAttributesA("C:\TradingLogs") = -1 then NoEntry = true;
if GetAppInfo(aiRealTimeCalc) = 1 and KillAttr = -1 and KillErr <> 2 and KillErr <> 3 then NoEntry = true;

if Time >= FirstEvalTime and Time <= FlatSignalTime and TodayOpen > 0 and PrevClose > 0 then
begin
    Slot = IntPortion((CurrentMinutes - FirstMinutes) / 15) + 1;

    if Slot >= 1 and Slot <= 22 then
    begin
        MoveToday = AbsValue(Close / TodayOpen - 1);
        SumMove = 0;
        EnoughData = false;

        if SlotCount[Slot] >= LookbackDays then
        begin
            EnoughData = true;
            for HistLoop = 1 to LookbackDays
            begin
                HistIndex = (Slot - 1) * 100 + HistLoop;
                SumMove = SumMove + MoveHistory[HistIndex];
            end;
            AvgMove = SumMove / LookbackDays;

            UpperBase = MaxList(TodayOpen, PrevClose);
            LowerBase = MinList(TodayOpen, PrevClose);
            UpperBand = UpperBase * (1 + AvgMove * VolatilityMultiplier);
            LowerBand = LowerBase * (1 - AvgMove * VolatilityMultiplier);
            LongTrail = MaxList(UpperBand, EMAValue);
            ShortTrail = MinList(LowerBand, EMAValue);

            if Time < FlatSignalTime then
            begin
                if MarketPosition = 0 and NoEntry = false and ContractsPerTrade >= 1 and TradesToday < MaxTradesPerDay
                   and StopsToday < MaxStopsPerDay then
                begin
                    if Time <= DayLastEntry then
                    begin
                        if LongTrade = true and Close > UpperBand and Close > Open then
                        begin
                            Buy ("NoiseLong") ContractsPerTrade contracts next bar at market;
                            TradesToday = TradesToday + 1;
                        end
                        else if ShortTrade = true and Close < LowerBand and Close < Open then
                        begin
                            SellShort ("NoiseShort") ContractsPerTrade contracts next bar at market;
                            TradesToday = TradesToday + 1;
                        end;
                    end;
                end;

                if MarketPosition = 1 then
                begin
                    if Time <= DayLastEntry and ShortTrade = true and Close < LowerBand and Close < Open then
                    begin
                        Sell ("LongRevExit") next bar at market;
                        if NoEntry = false and ContractsPerTrade >= 1 and TradesToday < MaxTradesPerDay and StopsToday < MaxStopsPerDay then
                        begin
                            SellShort ("RevToShort") ContractsPerTrade contracts next bar at market;
                            TradesToday = TradesToday + 1;
                        end;
                    end
                    else if Close < LongTrail then
                    begin
                        Sell ("LongEMAExit") next bar at market;
                    end;
                end;

                if MarketPosition = -1 then
                begin
                    if Time <= DayLastEntry and LongTrade = true and Close > UpperBand and Close > Open then
                    begin
                        BuyToCover ("ShortRevExit") next bar at market;
                        if NoEntry = false and ContractsPerTrade >= 1 and TradesToday < MaxTradesPerDay and StopsToday < MaxStopsPerDay then
                        begin
                            Buy ("RevToLong") ContractsPerTrade contracts next bar at market;
                            TradesToday = TradesToday + 1;
                        end;
                    end
                    else if Close > ShortTrail then
                    begin
                        BuyToCover ("ShortEMAExit") next bar at market;
                    end;
                end;
            end;
        end;

        for ShiftLoop = 100 downto 2
        begin
            MoveHistory[(Slot - 1) * 100 + ShiftLoop] = MoveHistory[(Slot - 1) * 100 + ShiftLoop - 1];
        end;
        MoveHistory[(Slot - 1) * 100 + 1] = MoveToday;
        if SlotCount[Slot] < 100 then
            SlotCount[Slot] = SlotCount[Slot] + 1;
    end;
end;

if Time = FlatSignalTime then
begin
    if MarketPosition = 1 then Sell ("FlatLong1430") next bar at market;
    if MarketPosition = -1 then BuyToCover ("FlatShort1430") next bar at market;
end;

// early-close flat (2026-11-05)
if IsEarlyClose and Time = EarlyFlattenTime then
begin
    if MarketPosition = 1 then Sell ("FlatLongEarly") next bar at market;
    if MarketPosition = -1 then BuyToCover ("FlatShortEarly") next bar at market;
end;

// RTH ANCHOR: remember the 15:00 close for tomorrow's PrevClose (after the day's logic)
if Time = RTHLastBarTime then
    LastRTHClose = Close;

// kill switch: flatten while C:\TradingLogs\KILL exists
if KillOn and MarketPosition = 1 then Sell ("KILL LX") all contracts next bar at market;
if KillOn and MarketPosition = -1 then BuyToCover ("KILL SX") all contracts next bar at market;

// heartbeat for the Chart Monitor / watchdog (Strategy_Heartbeat_Snippet.txt)
if HeartbeatTag <> "" and GetAppInfo(aiRealTimeCalc) = 1 and LastBarOnChart then
begin
    HB_Key = CurrentDate * 10000 + CurrentTime;
    if HB_Key <> HB_Last then
    begin
        HB_Day = NumToStr(CurrentDate + 19000000, 0);
        FileAppend("C:\TradingLogs\" + HB_Day + ".csv", "SIG," + HeartbeatTag + "," + GetSymbolName + "," + HB_Ver + ","
            + NumToStr(MarketPosition * CurrentContracts, 0) + ",,,,,,,," + HB_Day + "," + NumToStr(CurrentTime, 0) + NewLine);
        HB_Last = HB_Key;
    end;
end;
