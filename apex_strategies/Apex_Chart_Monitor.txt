// Apex Chart Monitor (2026-10-29) - SIGNAL, places NO orders. Live guard step 1 (live/vps/LIVE_GUARD_PLAN.md).
// Writes one ACCT row every MonitorMins to the same daily file as the Apex Trade Logger (C:\TradingLogs\YYYYMMDD.csv):
// the account's balance as Rithmic reports it, whether auto-trading is on for this chart's strategy, and how old the
// chart's last bar is. Claude's 16:40 CT job and the VPS watchdog read it: real balances on the dashboard, an alarm when
// auto-trading is off on a chart that should trade, an alarm when the data feed has gone quiet.
//
// How to use (outside market hours, once per live chart):
//   1. PowerLanguage Editor > New > Signal, name "Apex Chart Monitor", paste this file, compile (F7).
//   2. On each live chart: Format Signals > Add > Apex Chart Monitor. Inputs: Tag = the chart's logger tag (VB-2, ONB150, ...),
//      AccountName = the account typed exactly (the one live/expected_setup.json gives the tag, e.g. APEX-681524-02 or -04). Leave the others.
//   3. Auto-trading off / on for that chart (starting Flat outside a trade window), as after any signal change.
// Columns (same 14 as the logger, so one reader handles both):
//   ACCT,tag,symbol,account,auto_trading(1/0),net_worth,equity,bar_age_sec,route_account,,,,bar_date,bar_time
// route_account (2026-11-01): GetAccountID = the account this chart's strategies actually send orders to (the Auto Trading
// gear), empty while auto-trading is off. 'account' is only the typed AccountName input (used for the balance). The
// watchdog alarms when route_account is not the account live/expected_setup.json assigns to the tag.
// NetWorth 0 for an account = MultiCharts doesn't get the balance from Rithmic: tell Claude (the dashboard then keeps the
// balance the user reports).

inputs: Folder("C:\TradingLogs\"), Tag("VB-2"), AccountName(""), MonitorMins(5);
variables: Acct(""), Route(""), NW(0), EQ(0), AutoOn(0), AgeSec(0), FileName(""), BarDay(""), Row(""), Slot(0), LastSlot(-1),
	MinOfDay(0);

// keep calculating when no tick arrives, so a dead feed still produces rows with a growing bar age
RecalcLastBarAfter(60);

if GetAppInfo(aiRealTimeCalc) = 1 and LastBarOnChart then begin
	MinOfDay = IntPortion(CurrentTime * 0.01) * 60 + Mod(CurrentTime, 100);
	Slot = CurrentDate * 10000 + IntPortion(MinOfDay / MaxList(MonitorMins, 1));
	if Slot <> LastSlot then begin
		LastSlot = Slot;
		Route = GetAccountID;
		Acct = AccountName;
		if Acct = "" then begin
			NW = 0; EQ = 0;
		end
		else begin
			NW = GetRTAccountNetWorth(Acct);
			EQ = GetRTAccountEquity(Acct);
		end;
		AutoOn = GetAppInfo(aiStrategyAuto);
		AgeSec = (ComputerDateTime - DateTime) * 86400;
		BarDay = NumToStr(CurrentDate + 19000000, 0);
		FileName = Folder + BarDay + ".csv";
		Row = "ACCT," + Tag + "," + GetSymbolName + "," + Acct + "," + NumToStr(AutoOn, 0) + "," + NumToStr(NW, 2) + ","
			+ NumToStr(EQ, 2) + "," + NumToStr(AgeSec, 0) + "," + Route + ",,,," + BarDay + "," + NumToStr(CurrentTime, 0);
		FileAppend(FileName, Row + NewLine);
	end;
end;
