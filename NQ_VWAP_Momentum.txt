{ NQ VWAP Momentum - MultiCharts PowerLanguage signal

  Apply to a 5-minute NQ / MNQ chart. All times are HHMM in the CHART's time
  zone and default to US Central (CME exchange time). Eastern equivalents are
  shown in brackets.

  VWAP
    Anchored at the 8:30 CT [9:30 ET] open and built from 15-minute bars. The 15-minute bars
    are assembled from the 5-minute bars, so the value matches a 15-minute
    chart's VWAP and only updates when each 15-minute bar completes.

  LONG setup                              SHORT setup
    1. Close above VWAP                     1. Close below VWAP
    2. VWAP rising over the last 15 min     2. VWAP falling over the last 15 min
    3. Price up >= 0.1% over the last hour  3. Price down >= 0.1% over the last hour
  LONG trigger: first red candle of a       SHORT trigger: first green candle of a
    pullback (red after a non-red candle)     bounce (green after a non-green candle)
    while the setup is true                   while the setup is true
  Entry is at the open of the next bar.

  EXITS (points): long 80 stop / 40 target, short 80 stop / 50 target.

  GUARDRAILS
    One position at a time, max 4 trades a day, max 2 losing trades a day,
    no entries before 9:00 CT [10:00 ET] or after 14:30 CT [15:30 ET],
    flat at 14:55 CT [15:55 ET]. }

Inputs:
	AnchorTime(830),        { VWAP anchor: 8:30 CT = 9:30 ET open }
	VwapBarMins(15),        { VWAP is built from bars of this many minutes }
	MomentumMins(60),       { price change lookback in minutes }
	MinChangePct(0.1),      { minimum % move over MomentumMins }
	StopPts(80),
	LongTargetPts(40),
	ShortTargetPts(50),
	NoTradeUntil(900),      { first allowed entry time: 9:00 CT = 10:00 ET }
	LastEntryTime(1430),    { no new positions at or after this time: 14:30 CT = 15:30 ET }
	FlattenTime(1455),      { close any open position at this time: 14:55 CT = 15:55 ET }
	MaxTradesPerDay(4),
	MaxLossesPerDay(2),
	NumContracts(1);

Variables:
	MomentumBars(0), BarVol(0), MinsFromAnchor(0),
	VwapActive(false), BktOpen(false), BktHigh(0), BktLow(0), BktClose(0), BktVol(0),
	CumPV(0), CumVol(0), VwapNow(0), VwapPrev(0), VwapCount(0),
	PctChange(0),
	LongSetup(false), ShortSetup(false), LongTrigger(false), ShortTrigger(false),
	DayStartTrades(0), CountedTrades(0), LastNetProfit(0), TradesToday(0), LossesToday(0),
	InEntryWindow(false), CanEnter(false), TargetPts(0);

{ ---------- Setup ---------- }
Once begin
	if BarType <> 1 or Mod(VwapBarMins, BarInterval) <> 0 then
		RaiseRunTimeError("Use a minute chart whose interval divides evenly into " + NumToStr(VwapBarMins, 0) + " (5 minutes recommended).");
	MomentumBars = MaxList(1, Round(MomentumMins / BarInterval, 0));
	TargetPts = LongTargetPts;
end;

{ ---------- 15-minute VWAP anchored at the open ---------- }
if BarType < 2 then BarVol = Ticks else BarVol = Volume;
MinsFromAnchor = TimeToMinutes(Time) - TimeToMinutes(AnchorTime);

{ Start a fresh VWAP on the first bar that closes after the anchor time }
if MinsFromAnchor > 0 and (Date <> Date[1] or TimeToMinutes(Time[1]) <= TimeToMinutes(AnchorTime)) then begin
	VwapActive = true;
	BktOpen = false;
	CumPV = 0;
	CumVol = 0;
	VwapNow = 0;
	VwapPrev = 0;
	VwapCount = 0;
end
else if MinsFromAnchor <= 0 then
	VwapActive = false;

if VwapActive then begin
	{ add this bar to the current 15-minute bar }
	if BktOpen = false then begin
		BktHigh = High;
		BktLow = Low;
		BktVol = 0;
		BktOpen = true;
	end;
	BktHigh = MaxList(BktHigh, High);
	BktLow = MinList(BktLow, Low);
	BktClose = Close;
	BktVol = BktVol + BarVol;

	{ 15-minute bar complete: fold it into the VWAP }
	if Mod(MinsFromAnchor, VwapBarMins) = 0 then begin
		CumPV = CumPV + (BktHigh + BktLow + BktClose) / 3 * BktVol;
		CumVol = CumVol + BktVol;
		if CumVol > 0 then begin
			VwapPrev = VwapNow;
			VwapNow = CumPV / CumVol;
			VwapCount = VwapCount + 1;
		end;
		BktOpen = false;
	end;
end;

{ ---------- Setups and triggers ---------- }
PctChange = 0;
if CurrentBar > MomentumBars and Close[MomentumBars] > 0 then
	PctChange = (Close / Close[MomentumBars] - 1) * 100;

{ VwapCount >= 2 means there are two completed 15-minute VWAP values to compare }
LongSetup = VwapActive and VwapCount >= 2
	and Close > VwapNow
	and VwapNow > VwapPrev
	and PctChange >= MinChangePct;

ShortSetup = VwapActive and VwapCount >= 2
	and Close < VwapNow
	and VwapNow < VwapPrev
	and PctChange <= -MinChangePct;

{ first red candle of a pullback toward VWAP / first green candle of a bounce toward VWAP }
LongTrigger = LongSetup and Close < Open and not (Close[1] < Open[1]);
ShortTrigger = ShortSetup and Close > Open and not (Close[1] > Open[1]);

{ ---------- Daily guardrail counters ---------- }
if Date <> Date[1] then begin
	DayStartTrades = TotalTrades;
	LossesToday = 0;
end;

{ TotalTrades counts closed trades; one position at a time means one new trade per change }
if TotalTrades > CountedTrades then begin
	if NetProfit < LastNetProfit then LossesToday = LossesToday + 1;
	CountedTrades = TotalTrades;
	LastNetProfit = NetProfit;
end;

TradesToday = TotalTrades - DayStartTrades;
if MarketPosition <> 0 then TradesToday = TradesToday + 1;

{ ---------- Entries ---------- }
{ Orders fill at the next bar's open, so a decision on the bar closing at 9:00 CT
  fills at 9:00 CT and the last decision (bar closing 14:25 CT) fills at 14:25 CT. }
InEntryWindow = Time >= NoTradeUntil and Time < LastEntryTime;

CanEnter = InEntryWindow
	and MarketPosition = 0
	and TradesToday < MaxTradesPerDay
	and LossesToday < MaxLossesPerDay;

if MarketPosition = 1 then TargetPts = LongTargetPts
else if MarketPosition = -1 then TargetPts = ShortTargetPts;

if CanEnter then begin
	if LongTrigger then begin
		TargetPts = LongTargetPts;
		Buy ("Long") NumContracts contracts next bar at market;
	end
	else if ShortTrigger then begin
		TargetPts = ShortTargetPts;
		SellShort ("Short") NumContracts contracts next bar at market;
	end;
end;

{ ---------- Exits ---------- }
{ Stop and target in points, per contract. Called every bar so they stay active. }
SetStopContract;
SetStopLoss(StopPts * BigPointValue);
SetProfitTarget(TargetPts * BigPointValue);

{ Flat at FlattenTime }
if Time >= FlattenTime and MarketPosition <> 0 then begin
	Sell ("EOD LX") this bar on close;
	BuyToCover ("EOD SX") this bar on close;
end;
