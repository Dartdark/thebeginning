{ ----- Signal generation -----

    ReactionDay = False;

    If MyIsAMC[idx] and Date[1] = MyEvtDate[idx] then
        ReactionDay = True;

    If not MyIsAMC[idx] and Date = MyEvtDate[idx] then
        ReactionDay = True;

    If ReactionDay then begin
        ReactionPct = (Close - Close[1]) / Close[1] * 100;

           and (ReactionPct > 0 or UseReactionFilter = False) then
            PendingSignal = 1;

           and (ReactionPct < 0 or UseReactionFilter = False) then

    end;

    { Time-based exit after HoldDays }
    If BarsHeld >= HoldDays then
        BuyToCover ("Short-TimeExit") next bar at market;

    { Hard stop loss at fixed % above entry }
    If UseHardStop then
        BuyToCover ("Short-HardStop") next bar at
            (EntryPrice * (1 + HardStopPct / 100)) stop;

    { Trailing stop, only after position is up by TrailActivationPct }
    If UseTrailingStop and
       (EntryPrice - LowSinceEntry) / EntryPrice * 100 >= TrailActivationPct then
        BuyToCover ("Short-TrailStop") next bar at
            (LowSinceEntry * (1 + TrailStopPct / 100)) stop;
end;
