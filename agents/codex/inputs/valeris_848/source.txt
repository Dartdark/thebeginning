        { i dati di ieri }
        AtrOk = NReg >= AtrGiorni;
        AtrIeri = WAtr;
        AdrOk = AdrValida and AdrOggi > 0;
        AdrIeri = AdrOggi;
        ChiIeri = ChiOggi;
        { il peso della taglia dinamica di Valeris }
        Peso = 1;
        if AdrOk and MedOk then begin
            Peso = Power( MedIeri / AdrIeri, TagliaE );
            if Peso < 1 / TagliaTetto then Peso = 1 / TagliaTetto;
            if Peso > TagliaTetto then Peso = TagliaTetto;
        end;
    end;
    { sicurezza: mai una posizione aperta in una seduta nuova (in Valeris non succede) }
    if MarketPosition > 0 then Sell ( "848 sic LX" ) this bar on close;
    if MarketPosition < 0 then BuyToCover ( "848 sic SX" ) this bar on close;
    if MarketPosition <> 0 then Print( "S848 AVVISO ", Date, " ", Time, " posizione aperta a inizio seduta: chiusa" );
end;

if NuovaSeduta or CurrentBar = 1 then begin
    if Time > 1630 then SessData = JulianToDate( DateToJulian( Date ) + 1 ) else SessData = Date;
    Festa = false;
    for jj = 0 to NFeste - 1 begin
        if Feste[jj] = SessData then Festa = true;
    end;
    Mezza = false;
    for jj = 0 to NMezze - 1 begin
        if Mezze[jj] = SessData then Mezza = true;
    end;
    Ha830 = false;
    Ha1200 = false;
    Ha1500 = false;
    PrimaReg = true;
    Fase = 0;
    Verso = 0;
    NTrade = 0;
    InPos = false;
    BarIn = -1;
    UscitaBar = -1;
    HaFi = false;
    FiBar = -1;
    { stop, target e pareggio della seduta in punti, dall'ADR 14 di ieri; livelli dei rientri; contratti }
    if AdrOk then begin
        StopPunti = StopAdr * AdrIeri;
        TargetPunti = TargetAdr * AdrIeri;
        PareggioPunti = PareggioAdr * AdrIeri;
    end
    else begin
        StopPunti = 0;
        TargetPunti = 0;
        PareggioPunti = 0;
    end;
    { la distanza dello stop: quella di Valeris, o arrotondata per eccesso ai tick interi (i prezzi stanno sui tick:
      lo stop arrotondato lontano dall'entrata e' toccato esattamente quando e' toccato quello di Valeris) }
    DistStop = StopPunti;
    if ArrotondaStop = 1 and StopPunti > 0 then begin
        NTick = IntPortion( StopPunti / TickPunti );
        if StopPunti / TickPunti > NTick then NTick = NTick + 1;
        DistStop = NTick * TickPunti;
    end;
    LevL = ChiIeri + KAtr * AtrIeri;
    LevS = ChiIeri - KAtr * AtrIeri;
    if TagliaDinamica = 1 then Contratti = MaxList( 1, IntPortion( NumContracts * Peso + 0.5 ) )
    else Contratti = NumContracts;
    if CurrentBar = 1 then Fase = 9;             { la prima seduta del grafico e' a meta': niente }
end;

{ --- la seduta regolare di oggi: barre 08:35-15:00, chiusura delle 15:00 o delle 12:00 --- }
if Time = OraBase then begin
    Ha830 = true;
    Chi830 = Close;
end;
if Time > 830 and Time <= 1500 then begin
    if PrimaReg then begin
        MaxReg = High;
        MinReg = Low;
        PrimaReg = false;
    end
    else begin
        if High > MaxReg then MaxReg = High;
        if Low < MinReg then MinReg = Low;
    end;
end;
if Time = 1200 then begin
    Ha1200 = true;
    Chi1200 = Close;
end;
if Time = 1500 then begin
    Ha1500 = true;
    Chi1500 = Close;
end;

{ --- la colonna della barra nella seduta (0 = 17:00, 1 = 17:05, 187 = 08:35, 264 = 15:00) --- }
Minuti = IntPortion( Time / 100 ) * 60 + Mod( Time, 100 );
Col = IntPortion( Mod( Minuti - 1020 + 1440, 1440 ) / 5 );

{ ==========================================================================
  2. SUPERTREND (10, 2) E ATR 14 DELLE BARRE come VectorBT (SUPERTREND.run e
  ATR.run), su tutte le barre da 5 minuti del grafico, anche di notte: veri
  range dalla seconda barra, medie di Wilder (alfa = 1/n) valide dalla n-esima;
  bande base (H+L)/2 +- 2 ATR; bande finali che si stringono; verso che cambia
  solo se la chiusura supera la banda opposta; linea = banda bassa se il verso
  e' su, banda alta se e' giu'.
  ========================================================================== }
if CurrentBar = 1 then begin
    StDir = 1;
    StHaUp = false;
    StHaLo = false;
    StN = 0;
    StHaW = false;
    StOk = false;
    AbHaW = false;
    AbOk = false;
end
else begin
    TrB = MaxList( AbsValue( High - Low ), AbsValue( High - CPrec ), AbsValue( Low - CPrec ) );
    StN = StN + 1;
    if StHaW then begin
        if StW <> TrB then StW = ( FSt * StW + AlfaSt * TrB ) / ( FSt + AlfaSt );
    end
    else begin
        StW = TrB;
        StHaW = true;
    end;
    if AbHaW then begin
        if AbW <> TrB then AbW = ( FAb * AbW + AlfaAb * TrB ) / ( FAb + AlfaAb );
    end
    else begin
        AbW = TrB;
        AbHaW = true;
    end;
    AbOk = StN >= AtrBarre;
    if StN >= StPeriodo then begin
        MedB = ( High + Low ) / 2;
        UpB = MedB + StMolt * StW;
        LoB = MedB - StMolt * StW;
        if StHaUp = false or UpB < StUp or CPrec > StUp then StUp = UpB;
        if StHaLo = false or LoB > StLo or CPrec < StLo then StLo = LoB;
        StHaUp = true;
        StHaLo = true;
        if Close > StUp then StDir = 1
        else if Close < StLo then StDir = -1;
        if StDir > 0 then StLinea = StLo else StLinea = StUp;
        StOk = true;
    end;
end;
CIeriBarra = CPrec;
CPrec = Close;

{ --- la fine dei rientri: 6 barre dopo quella delle 15:00, mai oltre la barra delle 15:40; LucidFlex --- }
if HaFi = false and Col >= ColUscita and Col <= 282 then begin
    HaFi = true;
    FiBar = CurrentBar;
end;
AlFr = ( HaFi and CurrentBar >= FiBar + Oltre ) or ( Col >= ColCap and Col <= 282 );
Lucid = Col >= ColLucid and Col <= 282;

{ ==========================================================================
  3. un'operazione chiusa dallo stop, dal pareggio o dal target dentro questa
  barra: si annota la barra d'uscita (la pausa dei rientri parte da qui)
  ========================================================================== }
if InPos and CurrentBar > BarIn and MarketPosition = 0 then begin
    InPos = false;
    UscitaBar = CurrentBar;
    if NTrade >= NMax then Fase = 9 else Fase = 3;
end;

{ ==========================================================================
  4. uscite a orario: la base alla chiusura della barra delle 15:00, i
  rientri alla fine dei rientri (15:30; 15:40 al massimo); LucidFlex
  ========================================================================== }
if InPos and MarketPosition <> 0 then begin
    if ( NTrade = 1 and Col >= ColUscita and Col <= 282 ) or ( NTrade >= 2 and AlFr ) or Lucid then begin
        if MarketPosition > 0 then Sell ( "848 fine LX" ) this bar on close;
        if MarketPosition < 0 then BuyToCover ( "848 fine SX" ) this bar on close;
        InPos = false;
        UscitaBar = CurrentBar;
        if NTrade >= NMax or Lucid then Fase = 9 else Fase = 3;
    end;
end;

{ ==========================================================================
  5. la base alla barra delle 08:35: il verso della prima candela, il filtro
  Supertrend, l'entrata alla chiusura
  ========================================================================== }
if Col = ColEntrata and Fase = 0 then begin
    Entra = false;
    if Festa = false and Mezza = false and Ha830 and AtrOk and AdrOk and StOk and AbOk and Lucid = false then begin
        Candela = Close - Chi830;
        Verso = 0;
        if Candela > 0 then Verso = VersoRegola;
        if Candela < 0 then Verso = -VersoRegola;
        if Verso <> 0 and AbsValue( Candela ) >= SogliaAtr * AtrIeri and AbW > 0 then begin
            Zeta = Verso * ( Close - StLinea ) / AbW;
            if Zeta >= FiltroMin and Zeta < FiltroMax then Entra = true;
        end;
    end;
    { Lato: 1 solo long, -1 solo short; una seduta dell'altro verso non entra (Fase 9: niente base e niente rientri) }
    if Lato <> 0 and Verso <> Lato then Entra = false;
    if Entra then begin
        if Verso > 0 then Buy ( "848 L" ) Contratti contracts this bar on close;
        if Verso < 0 then SellShort ( "848 S" ) Contratti contracts this bar on close;
        Fase = 2;
        InPos = true;
        BarIn = CurrentBar;
        NTrade = 1;
        Migliore = Close;
        BeOk = false;
    end
    else Fase = 9;
end;

{ ==========================================================================
  6. i rientri: la chiusura passa il livello chiusura di ieri -/+ 0,25 ATR 14
  (la barra prima dalla parte di prima, questa dall'altra), nello stesso verso
  ========================================================================== }
if Fase = 3 and MarketPosition = 0 and InPos = false then begin
    if AlFr or Lucid then Fase = 9
    else if CurrentBar >= UscitaBar + 1 + Pausa then begin
        if Verso > 0 then Lev = LevL else Lev = LevS;
        Entra = false;
        if Verso > 0 and Close > Lev and CIeriBarra <= Lev then Entra = true;
        if Verso < 0 and Close < Lev and CIeriBarra >= Lev then Entra = true;
        if Entra then begin
            if Verso > 0 then Buy ( "848 RL" ) Contratti contracts this bar on close;
            if Verso < 0 then SellShort ( "848 RS" ) Contratti contracts this bar on close;
            Fase = 4;
            InPos = true;
            BarIn = CurrentBar;
            NTrade = NTrade + 1;
            Migliore = Close;
            BeOk = false;
        end;
    end;
end;

{ ==========================================================================
  7. il pareggio: il prezzo migliore dalle barre DOPO quella d'entrata; a 0,5
  ADR di guadagno lo stop va al prezzo d'entrata, dalla barra dopo
  ========================================================================== }
if InPos and MarketPosition <> 0 and CurrentBar > BarIn then begin
    if Verso > 0 and High > Migliore then Migliore = High;
    if Verso < 0 and Low < Migliore then Migliore = Low;
    if PareggioPunti > 0 and ( Migliore - EntryPrice ) * Verso >= PareggioPunti then BeOk = true;
    if BeOk then begin
        if MarketPosition > 0 then Sell ( "848 BE LX" ) next bar at EntryPrice stop;
        if MarketPosition < 0 then BuyToCover ( "848 BE SX" ) next bar at EntryPrice stop;
    end;
end;

{ ==========================================================================
  STOP E TARGET di MultiCharts, per contratto, dalla barra dopo l'entrata:
  0,1 ADR (ai tick interi con ArrotondaStop = 1) e 5 ADR 14 di ieri in punti
  (SetStopLoss e SetProfitTarget vogliono dollari per contratto: punti x
  BigPointValue, cioe' 2 $ a punto sul micro)
  ========================================================================== }
SetStopContract;
if DistStop > 0 then SetStopLoss( DistStop * BigPointValue );
if TargetPunti > 0 then SetProfitTarget( TargetPunti * BigPointValue );
