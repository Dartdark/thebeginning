## Reproduction of the main independent arithmetic/calendar checks

Bundled fixtures: `/workspace/venvs/thebeginning/bin/python -B analysis/codex_hunt_v2_vb_cov_features.py --fixtures`.
The following core excerpt reads only code and invented in-memory observations; it imports no actual
calendar callback. The file-pipeline tests use the same public functions with temporary ROOT/OUT and
an artificial `codex_s03_calendar` module, guarding `compute_features` so any failed-source call raises.

```python
import sys, random
sys.dont_write_bytecode = True
sys.path.insert(0, 'analysis')
import codex_hunt_v2_vb_cov_features as m
from fractions import Fraction as F
from datetime import date, timedelta
rng = random.Random(90261027)
for trial in range(40):
    x = [F(rng.randrange(-10000,10001),100) for _ in range(252)]
    b = [F(rng.randrange(-90000,90001),100) for _ in range(252)]
    if trial % 3 == 0: b = [-v for v in x]
    elif trial % 3 == 1: b = x.copy()
    rows = [m.invented_row(date(2000,1,1)+timedelta(days=i),a,z,i)
            for i,(a,z) in enumerate(zip(x,b))]
    r = m.risk_from_window(rows)
    xm,bm = sum(x)/252,sum(b)/252
    var = sum((z-xm)**2 for z in x)/251
    cov = sum((a-xm)*(z-bm) for a,z in zip(x,b))/251
    assert F(r['sample_parent_variance_exact']) == var
    assert F(r['sample_covariance_book_parent_exact']) == cov
    assert F(r['sample_delta_variance_exact']) == 2*cov+var
    assert r['qty_extra'] == int(var>0 and 2*cov+var<0)
normal = {f'{y}-01-02' for y in range(2018,2023)}
early = {'2020-01-03'}
fake = lambda d: dict(kind='normal' if d in normal else
    'early' if d in early else 'holiday',reason='ARTIFICIAL_ONLY')
rows = [m.invented_row(date(y,1,d),1,-1,y*10+d)
        for y in range(2018,2023) for d in (2,3)]
a = m.calendar_audit(rows,fake)
assert a['passed'] and a['expected_normal_or_early_dates'] == 6
assert len(a['extra_observed_non_normal_or_early_dates']) == 4
for missing in ('2019-01-02','2020-01-03'):
    a = m.calendar_audit([r for r in rows if r['date'].isoformat()!=missing],fake)
    assert not a['passed'] and a['missing_expected_normal_or_early_dates']==[missing]
```
